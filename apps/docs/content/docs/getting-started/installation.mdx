## Quick Start

```bash
bun create questpie my-app
cd my-app
```

This scaffolds a project with the file convention layout, runtime config, and a starter collection.

## Manual Setup

<Steps>
<Step title="Install packages">

```bash
bun add questpie @questpie/hono
```

For the admin panel (optional):

```bash
bun add @questpie/admin
```

</Step>
<Step title="Create runtime config">

```ts title="src/questpie/server/questpie.config.ts"
import { runtimeConfig } from "questpie/app";
export default runtimeConfig({
	app: {
		url: process.env.APP_URL || "http://localhost:3000",
	},
	db: {
		url: process.env.DATABASE_URL || "postgres://localhost/myapp",
	},
	secret: process.env.APP_SECRET || "change-me-in-production",
});
```

</Step>
<Step title="Create CLI config">

```ts title="questpie.config.ts"
export { default } from "./src/questpie/server/questpie.config";
```

The CLI resolves this file at the project root to find your config.

</Step>
<Step title="Create your first collection">

```ts title="src/questpie/server/collections/posts.ts"
import { collection } from "#questpie/factories";

export default collection("posts").fields(({ f }) => ({
	title: f.text().required(),
	body: f.textarea(),
	status: f.select([
		{ value: "draft", label: "Draft" },
		{ value: "published", label: "Published" },
	]),
}));
```

</Step>
<Step title="Run codegen">

```bash
bunx questpie generate
```

This scans your file convention directories, discovers collections/routes/jobs, and generates:

- `src/questpie/server/.generated/` — app instance, types, module augmentation

</Step>
<Step title="Push schema to database">

```bash
bunx questpie push
```

This syncs your Drizzle schema to the database without creating migration files. Use `questpie migrate:generate` for production migrations.

</Step>
<Step title="Start development">

```bash
bun dev
```

</Step>
</Steps>

## With Admin Panel

To add the admin interface, install `@questpie/admin` and add the admin module:

Add the admin module to your modules:

```ts title="src/questpie/server/modules.ts"
import { adminModule } from "@questpie/admin/modules/admin";
export default [adminModule] as const;
```

Create `src/questpie/server/config/admin.ts` for sidebar, dashboard, branding, and locale, then re-run codegen:

```bash
bunx questpie generate
```

## Adapters

QUESTPIE needs an HTTP adapter to serve your API. Choose one:

| Adapter        | Package            | Use case               |
| -------------- | ------------------ | ---------------------- |
| Hono           | `@questpie/hono`   | General purpose, fast  |
| Elysia         | `@questpie/elysia` | Bun-native, validation |
| Next.js        | `@questpie/next`   | Next.js API routes     |
| TanStack Start | (none)             | Built-in fetch handler |

<Tabs items={["TanStack Start", "Hono", "Elysia", "Next.js"]}>
  <Tab value="TanStack Start">
    ```ts title="src/routes/api/$.ts"
    import { createAPIFileRoute } from "@tanstack/react-start/api";
    import { createFetchHandler } from "questpie/http";
    import { app } from "#questpie";

    const handler = createFetchHandler(app, { basePath: "/api" });

    export const Route = createAPIFileRoute("/api/$")({
      GET: ({ request }) => handler(request),
      POST: ({ request }) => handler(request),
      PATCH: ({ request }) => handler(request),
      DELETE: ({ request }) => handler(request),
    });
    ```
  </Tab>
  <Tab value="Hono">
    ```bash
    bun add @questpie/hono hono
    ```

    ```ts title="src/index.ts"
    import { Hono } from "hono";
    import { questpieHono } from "@questpie/hono/server";
    import { app } from "#questpie";

    const server = new Hono();
    server.route("/api", questpieHono(app));

    export default server;
    ```
  </Tab>
  <Tab value="Elysia">
    ```bash
    bun add @questpie/elysia elysia
    ```

    ```ts title="src/index.ts"
    import { Elysia } from "elysia";
    import { questpieElysia } from "@questpie/elysia/server";
    import { app } from "#questpie";

    const server = new Elysia()
      .use(questpieElysia(app, { basePath: "/api" }))
      .listen(3000);
    ```
  </Tab>
  <Tab value="Next.js">
    ```bash
    bun add @questpie/next
    ```

    ```ts title="app/api/[...slug]/route.ts"
    import { questpieNextRouteHandlers } from "@questpie/next";
    import { app } from "#questpie";

    export const { GET, POST, PATCH, DELETE } = questpieNextRouteHandlers(app, {
      basePath: "/api",
    });
    ```
  </Tab>
</Tabs>

## Environment Variables

| Variable       | Required | Description                  |
| -------------- | -------- | ---------------------------- |
| `DATABASE_URL` | Yes      | PostgreSQL connection string |
| `APP_URL`      | Yes      | Public URL of your app       |
| `APP_SECRET`   | Yes      | Secret for auth sessions     |
| `SMTP_HOST`    | No       | Email SMTP host              |
| `SMTP_PORT`    | No       | Email SMTP port              |
