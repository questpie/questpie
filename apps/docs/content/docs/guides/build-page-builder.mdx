---
title: Build a Page Builder with Blocks
description: Define blocks and use them to create a flexible page builder.
---

Blocks let editors compose pages from reusable content sections without writing code. Define block types server-side with fields and admin metadata, then render them client-side with React components.

## 1. Define the blocks

Create block files in `src/questpie/server/blocks/`. Each block declares its fields and how it appears in the admin picker.

```ts title="src/questpie/server/blocks/hero.ts"
import { block } from "#questpie/factories";

export const heroBlock = block("hero")
  .admin(({ c }) => ({
    label: { en: "Hero Section" },
    icon: c.icon("ph:image"),
    category: "sections",
  }))
  .fields(({ f }) => ({
    title: f.text().required().localized(),
    subtitle: f.textarea().localized(),
    backgroundImage: f.upload(),
    ctaText: f.text().localized(),
    ctaLink: f.text(),
  }))
  .prefetch({ with: { backgroundImage: true } });
```

```ts title="src/questpie/server/blocks/cta.ts"
import { block } from "#questpie/factories";

export const ctaBlock = block("cta")
  .admin(({ c }) => ({
    label: { en: "Call to Action" },
    icon: c.icon("ph:megaphone"),
    category: "sections",
  }))
  .fields(({ f }) => ({
    title: f.text().required().localized(),
    description: f.textarea().localized(),
    buttonText: f.text().localized(),
    buttonLink: f.text(),
    variant: f
      .select([
        { value: "highlight", label: "Highlight" },
        { value: "dark", label: "Dark" },
        { value: "light", label: "Light" },
      ])
      .default("highlight"),
  }));
```

## 2. Add a blocks field to a collection

Use `f.blocks()` on any collection field to enable the block editor in the admin.

```ts title="src/questpie/server/collections/pages.ts"
import { collection } from "#questpie/factories";

export const pages = collection("pages")
  .fields(({ f }) => ({
    title: f.text(255).required().localized(),
    slug: f.text(255).required().inputOptional(),
    content: f.blocks().localized(),
  }))
  .admin(({ c }) => ({
    label: { en: "Pages" },
    icon: c.icon("ph:article"),
  }));
```

## 3. Write the block renderers

Create a React component for each block. The `BlockProps<"blockName">` type is generated from your server block definition and provides fully typed `values` and `data`.

```tsx title="src/questpie/admin/blocks/hero.tsx"
import { PreviewField } from "@questpie/admin/client";
import type { BlockProps } from "../.generated/client";

export function HeroRenderer({ values, data }: BlockProps<"hero">) {
  return (
    <section
      className="relative flex min-h-[60vh] items-center"
      style={{
        backgroundImage: data?.backgroundImage?.url
          ? `url(${data.backgroundImage.url})`
          : undefined,
      }}
    >
      <div className="relative z-10 container text-center">
        <PreviewField field="title" editable="text" as="h1">
          {values.title}
        </PreviewField>
        {values.subtitle && (
          <PreviewField field="subtitle" editable="textarea" as="p">
            {values.subtitle}
          </PreviewField>
        )}
        {values.ctaText && (
          <a href={values.ctaLink}>{values.ctaText}</a>
        )}
      </div>
    </section>
  );
}
```

```tsx title="src/questpie/admin/blocks/cta.tsx"
import { PreviewField } from "@questpie/admin/client";
import type { BlockProps } from "../.generated/client";

export function CTARenderer({ values }: BlockProps<"cta">) {
  return (
    <section className="py-20 text-center">
      <PreviewField field="title" editable="text" as="h2">
        {values.title}
      </PreviewField>
      {values.description && (
        <PreviewField field="description" editable="textarea" as="p">
          {values.description}
        </PreviewField>
      )}
      {values.buttonText && (
        <a href={values.buttonLink}>{values.buttonText}</a>
      )}
    </section>
  );
}
```

## 4. Render blocks on the frontend

Use `BlockRenderer` with the generated `admin.blocks` map. Wrap with `PreviewProvider` and `useCollectionPreview` to enable Live Preview.

```tsx title="src/components/PageRenderer.tsx"
import { useRouter } from "@tanstack/react-router";
import {
  type BlockContent,
  BlockRenderer,
  PreviewProvider,
  useCollectionPreview,
} from "@questpie/admin/client";
import admin from "@/questpie/admin/.generated/client";

export function PageRenderer({ page }) {
  const router = useRouter();
  const preview = useCollectionPreview({
    initialData: page,
    onRefresh: () => router.invalidate(),
  });

  return (
    <PreviewProvider preview={preview}>
      <BlockRenderer
        content={preview.data.content as BlockContent}
        renderers={admin.blocks}
        data={preview.data.content?._data}
        selectedBlockId={preview.selectedBlockId}
        onBlockClick={
          preview.isPreviewMode ? preview.handleBlockClick : undefined
        }
      />
    </PreviewProvider>
  );
}
```

<Callout type="info">
`BlockProps<"hero">` and `admin.blocks` are generated by `questpie codegen`. Run codegen after adding new block files so the types stay in sync.
</Callout>
